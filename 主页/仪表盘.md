## 3 🔍 智能检索

### 3.1 🏷️ 热门标签
```dataviewjs
// 获取热门标签
const tags = {};
for (let page of dv.pages()) {
  if (page.file.tags) {
    for (let tag of page.file.tags) {
      if (!tag.startsWith("project/") && !tag.startsWith("主页/")) {
        tags[tag] = (tags[tag] || 0) + 1;
      }
    }
  }
}

// 排序并显示前20个
const sortedTags = Object.entries(tags)
  .sort((a, b) => b[1] - a[1])
  .slice(0, 20);

dv.paragraph("**Top 20 热门标签：**\n");
const tagList = sortedTags.map(([tag, count]) => 
  `#${tag} (${count})`
).join(" · ");

dv.paragraph(tagList);
```
